\newcommand{\columnsum}[3]{
	\pgfplotstableread[col sep=semicolon, ignore chars={\#,\%,\_,\>,\<}]{#1}\loadedtable
	\pgfplotstablegetrowsof{\loadedtable}
	\pgfmathtruncatemacro{\numrows}{\pgfplotsretval}
	\pgfmathsetmacro{#3}{0}
	 \pgfmathsetmacro{\localsum}{0}

    \pgfplotstableforeachcolumnelement{[index] #2}\of\loadedtable\as\cell{%
        \pgfmathparse{\localsum+\cell}%
        \xdef\localsum{\pgfmathresult}%
    }

    \global\let#3\localsum
}

\newcommand{\columnmean}[3]{
	\pgfplotstableread[col sep=semicolon, ignore chars={\#,\%,\_,\>,\<}]{#1}\loadedtable
	\pgfplotstablegetrowsof{\loadedtable}
	\pgfmathtruncatemacro{\numrows}{\pgfplotsretval}
	\pgfmathsetmacro{#3}{0}
	\pgfmathsetmacro{\localmean}{0}

	\pgfplotstableforeachcolumnelement{[index] #2}\of\loadedtable\as\cell{%
		\pgfmathparse{\localmean+\cell}%
		\xdef\localmean{\pgfmathresult}%
	}
	\pgfmathsetmacro{#3}{\localmean/\numrows}
}
